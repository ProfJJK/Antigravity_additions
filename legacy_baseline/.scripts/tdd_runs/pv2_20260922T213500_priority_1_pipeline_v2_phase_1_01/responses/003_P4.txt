"""Phase 1 infrastructure tests for Pipeline v2 (Task 1.01).

[T1] test_directory_scaffold_exists  -> AC1 (AC-10, FR-2.8)
[T2] test_v1_isolation_guard         -> AC2 (AC-29, INV-4)

All assertions run against the real filesystem and, when available, the
real git worktree. No mocks, no skips. The legacy SQLite database is only
inspected via ``stat``; it is never opened, so the check cannot itself
perturb the mtime it verifies.
"""

from __future__ import annotations

import hashlib
import json
import subprocess
import sys
from pathlib import Path

import pytest

REPO_ROOT = Path(r"D:\__CoChem")
AGENTIC_V2_ROOT = REPO_ROOT / "__agentic" / "v2"

GITKEEP_PATH = AGENTIC_V2_ROOT / "data" / ".gitkeep"
INIT_PATH = AGENTIC_V2_ROOT / "tests" / "__init__.py"

LEGACY_SCRIPTS_DIR = REPO_ROOT / ".scripts"
LEGACY_DB_PATH = REPO_ROOT / "cochem_kanban.db"

# Persisted isolation baseline owned by this suite. It lives beside this
# module inside __agentic/v2/tests, never under any legacy v1 path.
BASELINE_PATH = Path(__file__).resolve().parent / "_phase1_v1_isolation_baseline.json"

CREATE_NO_WINDOW = 0x08000000


def _subprocess_kwargs() -> dict:
    kwargs: dict = {"capture_output": True, "text": True, "encoding": "utf-8"}
    if sys.platform.startswith("win"):
        kwargs["creationflags"] = getattr(subprocess, "CREATE_NO_WINDOW", CREATE_NO_WINDOW)
    return kwargs


def _hash_tree(directory: Path) -> str:
    """SHA-256 over relative path, size and mtime_ns of every file below directory."""
    hasher = hashlib.sha256()
    for path in sorted(directory.rglob("*")):
        if path.is_file():
            st = path.stat()
            hasher.update(path.relative_to(directory).as_posix().encode("utf-8"))
            hasher.update(b"\0")
            hasher.update(str(st.st_size).encode("utf-8"))
            hasher.update(b"\0")
            hasher.update(str(st.st_mtime_ns).encode("utf-8"))
            hasher.update(b"\n")
    return hasher.hexdigest()


def _git_status_porcelain(pathspec: str) -> subprocess.CompletedProcess:
    return subprocess.run(
        ["git", "status", "--porcelain", "--untracked-files=all", "--", pathspec],
        cwd=str(REPO_ROOT),
        **_subprocess_kwargs(),
    )


def test_directory_scaffold_exists() -> None:
    """[T1] Bind anchor and test package marker exist as readable regular files."""
    assert GITKEEP_PATH.is_file(), f"Missing host mount bind anchor: {GITKEEP_PATH}"
    assert INIT_PATH.is_file(), f"Missing test package marker: {INIT_PATH}"

    # The anchor's parent is the exact directory the compose volume binds to.
    assert GITKEEP_PATH.parent.is_dir()
    assert GITKEEP_PATH.parent.as_posix().endswith("__agentic/v2/data")

    gitkeep_text = GITKEEP_PATH.read_text(encoding="utf-8")
    init_text = INIT_PATH.read_text(encoding="utf-8")

    assert isinstance(gitkeep_text, str)
    assert init_text.strip(), "__init__.py must contain a real module docstring, not be empty."
    # The package marker must be valid, importable Python source.
    compile(init_text, str(INIT_PATH), "exec")


def test_v1_isolation_guard() -> None:
    """[T2] .scripts/ has no git modifications and cochem_kanban.db mtime is unchanged."""
    assert LEGACY_DB_PATH.is_file(), f"Legacy database missing at {LEGACY_DB_PATH}"
    assert LEGACY_SCRIPTS_DIR.is_dir(), f"Legacy scripts dir missing at {LEGACY_SCRIPTS_DIR}"

    if (REPO_ROOT / ".git").exists():
        result = _git_status_porcelain(".scripts")
        assert result.returncode == 0, f"git status failed ({result.returncode}): {result.stderr!r}"
        assert result.stdout.strip() == "", (
            f"Modifications detected under .scripts/: {result.stdout!r}"
        )

    current = {
        "db_mtime_ns": LEGACY_DB_PATH.stat().st_mtime_ns,
        "scripts_hash": _hash_tree(LEGACY_SCRIPTS_DIR),
    }

    if not BASELINE_PATH.is_file():
        BASELINE_PATH.write_text(json.dumps(current, indent=2, sort_keys=True), encoding="utf-8")
        baseline = json.loads(BASELINE_PATH.read_text(encoding="utf-8"))
    else:
        baseline = json.loads(BASELINE_PATH.read_text(encoding="utf-8"))

    assert current["db_mtime_ns"] == baseline["db_mtime_ns"], (
        f"cochem_kanban.db mtime drifted: baseline={baseline['db_mtime_ns']} "
        f"observed={current['db_mtime_ns']} (INV-4 / AC-29 violation)."
    )
    assert current["scripts_hash"] == baseline["scripts_hash"], (
        ".scripts/ tree changed relative to recorded baseline (INV-4 / AC-29 violation)."
    )


if __name__ == "__main__":
    raise SystemExit(pytest.main([__file__, "-v"]))
