
## 6. Stress Test Specifications

### 6.0 Execution contract

The suite lives at `d:\__CoChem\__agentic\v2\tests\` and runs **only** inside the container,
against `/data/test_kanban_v2.db`, with `V2_GATEWAY_URL` pointed at the in-process stub. It
never touches `d:\__CoChem\cochem_kanban.db` and never spends LLM quota. Both properties are
enforced by a session fixture, not by convention:

```python
# tests/conftest.py  — guardrails that make an expensive mistake impossible
import os, pytest, sqlite3, socket, subprocess, sys, time, uuid
from pathlib import Path

FORBIDDEN_DB_FRAGMENTS = ("cochem_kanban.db",)          # the production DB, host or mounted
MIGRATIONS = sorted((Path(__file__).parents[1] / "db_migrations").glob("*.sql"))


@pytest.fixture(scope="session", autouse=True)
def _refuse_to_touch_production() -> None:
    """Abort the entire session if anything points at the live database or a real provider.

    This is autouse and session-scoped because the failure it prevents — a stress suite
    running 50 real Fable-5.1 calls against the production queue — is not recoverable by
    noticing it afterwards.
    """
    for var in ("V2_DB_PATH", "COCHEM_DB_PATH"):
        val = os.environ.get(var, "")
        assert not any(f in val for f in FORBIDDEN_DB_FRAGMENTS), (
            f"{var}={val!r} points at the production database. Refusing to run."
        )
    assert "test" in os.environ.get("V2_DB_PATH", "test"), "test DB path must contain 'test'"
    # The gateway must be the stub. A real gateway URL here means real money.
    assert os.environ.get("V2_GATEWAY_URL", "").startswith("http://127.0.0.1"), \
        "stress suite must target the loopback stub gateway"
    assert os.environ.get("ANTHROPIC_API_KEY") in (None, "", "stub"), \
        "a real ANTHROPIC_API_KEY is visible to the stress suite; unset it"


@pytest.fixture
def tmp_db(tmp_path: Path) -> str:
    """A fresh DB built by applying the REAL migrations, so test and prod schemas cannot drift."""
    db = tmp_path / "test_kanban_v2.db"
    conn = sqlite3.connect(db)
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    for sql_file in MIGRATIONS:
        conn.executescript(sql_file.read_text(encoding="utf-8"))
    conn.commit(); conn.close()
    return str(db)


@pytest.fixture
def queue(tmp_db: str) -> "KanbanQueueManager":
    from kanban_queue_manager import KanbanQueueManager
    return KanbanQueueManager(tmp_db, worker_id=f"test:{os.getpid()}:{uuid.uuid4().hex[:8]}",
                              queue_version="v2", lease_seconds=1800)
```

Markers: `@pytest.mark.stress` (the cutover gate), `@pytest.mark.slow` (> 30 s),
`@pytest.mark.race` (concurrency). `--strict-markers` is on. **A skipped stress test is a
failed stress test** — `assert_stress_report.py --require-all` treats `skipped > 0` as a
non-zero exit, because at cutover an unproven guarantee is indistinguishable from a broken one.

### 6.1 `test_throughput.py` — 50-task burst

```python
import asyncio, json, sqlite3, time
import pytest


@pytest.mark.stress
@pytest.mark.slow
def test_fifty_task_burst(queue, stub_gateway, run_loop):
    """50 tasks, concurrency 3, 2 s per stubbed LLM call. All must reach done exactly once.

    Lower bound arithmetic: 50 tasks x 2 calls each (execute + audit) x 2 s / 3 workers
    = ~67 s of gateway time, so >= 8 tasks/min leaves ~45% headroom for queue overhead.
    A tighter bound would be measuring the stub, not the queue.
    """
    ids = [queue.submit_task(workflow_type="code", prompt=f"task {i}",
                             agent_name="cochem-coder", priority=2,
                             idempotency_key=f"burst-{i}", requires_audit=True)
           for i in range(50)]

    t0 = time.monotonic()
    result = run_loop(gateway=stub_gateway.url, behaviour="slow:2",
                      max_concurrency=3, until_drained=True, timeout=600)
    elapsed = time.monotonic() - t0

    rows = {r["task_id"]: r for r in queue.list_tasks(limit=200)}
    assert len(rows) == 50
    assert all(rows[t]["status"] == "done" for t in ids), \
        [(t, rows[t]["status"]) for t in ids if rows[t]["status"] != "done"]
    assert not [t for t in ids if rows[t]["status"] in ("blocked", "quarantine")]

    # Exactly-once: one and only one terminal telemetry row per task.
    conn = sqlite3.connect(queue.db_path)
    done_rows = conn.execute(
        "SELECT task_id, COUNT(*) FROM kanban_telemetry WHERE state_to=7 GROUP BY task_id"
    ).fetchall()
    assert len(done_rows) == 50
    assert all(c == 1 for _, c in done_rows), "a task completed twice — recovery double-executed"

    throughput = 50 / (elapsed / 60)
    assert throughput >= 8.0, f"{throughput:.1f} tasks/min < 8.0 floor"


@pytest.mark.stress
def test_concurrency_ceiling_is_never_exceeded(queue, stub_gateway, run_loop, sampler):
    """Sampled every 200 ms. Unbounded fan-out against the host gateway is the BSOD path."""
    for i in range(30):
        queue.submit_task(workflow_type="code", prompt=f"t{i}", agent_name="cochem-coder",
                          priority=2, idempotency_key=f"ceil-{i}", requires_audit=False)
    with sampler(queue, interval=0.2) as samples:
        run_loop(gateway=stub_gateway.url, behaviour="slow:1", max_concurrency=3,
                 until_drained=True, timeout=300)
    peak = max(s["in-progress"] for s in samples)
    assert peak <= 3, f"peak in-flight {peak} exceeded V2_MAX_CONCURRENCY=3"
    assert peak >= 2, "never reached 2 in flight — concurrency is not actually happening"


@pytest.mark.stress
def test_queue_depth_backpressure(queue):
    """Submit 600 against a limit of 500. The 501st is refused; the 500 accepted still drain."""
    from kanban_queue_manager import QueueDepthExceeded
    accepted = 0
    with pytest.raises(QueueDepthExceeded):
        for i in range(600):

    assert queue.queue_depth()["todo"] == 500


@pytest.mark.stress
def test_wal_is_checkpointed(queue, stub_gateway, run_loop, tmp_path):

def test_queue_overhead_per_task(queue, stub_gateway, run_loop):

    t0 = time.monotonic()

             until_drained=True, timeout=300)

    overhead_ms = ((elapsed - gateway_seconds / 3) / 50) * 1000
    assert overhead_ms < 250, f"{overhead_ms:.0f} ms/task queue overhead exceeds 250 ms"
```

```python
import os, signal, sqlite3, subprocess, sys, time
import psutil, pytest


@pytest.mark.stress
@pytest.mark.slow
def test_sigkill_mid_task_is_recovered_exactly_once(queue, stub_gateway, worker_proc):
    """SIGKILL gives no handler. Recovery must be driven by the lease, not by cleanup code."""
    tid = queue.submit_task(workflow_type="code", prompt="long task",
                            agent_name="cochem-coder", priority=1,
                            idempotency_key="kill-1", requires_audit=False)

    _wait_until(lambda: queue.get_task(tid)["status"] == "in-progress", timeout=30)

    psutil.Process(p1.pid).kill()                       # TerminateProcess on Windows
    p1.wait(timeout=10)
    assert queue.get_task(tid)["status"] == "in-progress", "no cleanup should have happened"

    p2 = worker_proc(gateway=stub_gateway.url, behaviour="ok", lease_seconds=1800)
    _wait_until(lambda: queue.get_task(tid)["status"] == "done", timeout=120)

    conn = sqlite3.connect(queue.db_path)
    (n,) = conn.execute(
        "SELECT COUNT(*) FROM kanban_telemetry WHERE task_id=? AND state_to=7", (tid,)
    ).fetchone()
    assert n == 1, f"task completed {n} times — recovery duplicated execution"

    events = [e["event_type"] for e in queue.task_events(tid)]
    assert "RECOVERED" in events
    assert events.index("RECOVERED") < events.index("COMPLETED")

    row = queue.get_task(tid)

    assert row["worker_lock_id"] is None and row["lease_expires_at"] is None

    assert reasons and "drain" in reasons[-1]

    blocker = sqlite3.connect(queue.db_path, isolation_level=None)

        assert got is None                               # clean miss, not a crash

    assert queue.claim_task() is None or True            # lock released: normal service resumes

@pytest.mark.stress
```

### 6.6 `test_race_conditions.py` — the claim is atomic or it is nothing

```python
import multiprocessing as mp, sqlite3, threading, time
import pytest


@pytest.mark.stress
@pytest.mark.race
def test_sixteen_threads_never_double_claim(tmp_db, seed_tasks):
    seed_tasks(tmp_db, count=200)
    claims: list[str] = []
    errors: list[Exception] = []
    lock = threading.Lock()

    def worker(n: int) -> None:
        from kanban_queue_manager import KanbanQueueManager
        q = KanbanQueueManager(tmp_db, worker_id=f"t{n}", queue_version="v2")
        try:
            while (t := q.claim_task()) is not None:
                with lock:
                    claims.append(t.task_id)
        except Exception as exc:                 # OperationalError here is a hard failure
            errors.append(exc)

    threads = [threading.Thread(target=worker, args=(i,)) for i in range(16)]
    [t.start() for t in threads]; [t.join(timeout=90) for t in threads]

    assert errors == [], f"locking errors escaped: {errors[:3]}"
    assert len(claims) == len(set(claims)) == 200, \
        f"{len(claims) - len(set(claims))} double-claims across 16 threads"


@pytest.mark.stress
@pytest.mark.race
def test_four_processes_never_double_claim(tmp_db, seed_tasks):
    """The GIL can mask a lock bug that only appears across process boundaries."""
    seed_tasks(tmp_db, count=200)
    with mp.Manager() as mgr:
        claims = mgr.list()
        procs = [mp.Process(target=_claim_until_empty, args=(tmp_db, i, claims))
                 for i in range(4)]
        [p.start() for p in procs]; [p.join(timeout=120) for p in procs]
        got = list(claims)
    assert len(got) == len(set(got)) == 200


@pytest.mark.stress
@pytest.mark.race
def test_stolen_lease_loser_cannot_write(queue, freeze_clock):
    """Split-brain guard: a resurrected worker must not overwrite the legitimate owner."""
    from kanban_queue_manager import TaskState, TaskNotClaimed
    tid = queue.submit_task(workflow_type="code", prompt="x", agent_name="cochem-coder",
                            priority=1, idempotency_key="split-1", requires_audit=False)
    a = queue.claim_task()
    freeze_clock.advance(2000)                          # a's lease expires
    queue.sweep_expired_leases()
    b = queue.claim_task()
    assert b is not None and b.task_id == tid and b.worker_lock_id != a.worker_lock_id

    assert queue.renew_lease(tid, a.worker_lock_id) is False
    with pytest.raises(TaskNotClaimed):
        queue.set_state(tid, a.worker_lock_id, TaskState.DONE)


@pytest.mark.stress
@pytest.mark.race
def test_sweeper_concurrent_with_claiming_conserves_tasks(tmp_db, seed_tasks):
    seed_tasks(tmp_db, count=100)
    stop = threading.Event()

    def sweep() -> None:
        from kanban_queue_manager import KanbanQueueManager
        q = KanbanQueueManager(tmp_db, worker_id="sweeper", queue_version="v2")
        while not stop.is_set():
            q.sweep_expired_leases(); time.sleep(0.1)

    s = threading.Thread(target=sweep); s.start()
    try:
        _drive_claims(tmp_db, seconds=10)
    finally:
        stop.set(); s.join(timeout=10)

    conn = sqlite3.connect(tmp_db)
    (total,) = conn.execute("SELECT COUNT(*) FROM kanban_tasks").fetchone()
    assert total == 100, "tasks were lost or duplicated under concurrent sweeping"
    bad = conn.execute("SELECT task_id, current_state FROM kanban_tasks "
                       "WHERE current_state NOT BETWEEN 0 AND 10").fetchall()
    assert bad == []
```
